# arch-tidy — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build the `arch-tidy` skill — a deterministic, read-only analysis script (`analyze.py`) plus an agent playbook (`SKILL.md`) that produces selective hierarchical folder docs (`_folder_context.md`) and proposes (never auto-applies) file moves.

**Architecture:** A pure-stdlib Python analyzer emits a structured `arch-report.json`; the agent, guided by `SKILL.md`, reasons over that report to propose moves (approval-gated) and to create/update `_folder_context.md` files bottom-up. The script never mutates the filesystem.

**Tech Stack:** Python 3 standard library only (no third-party deps). Tests use stdlib `unittest`. Markdown for `SKILL.md` and the schema reference.

**Source spec:** `docs/specs/2026-07-09-arch-tidy-design.md`.

## Global Constraints

- **Skill home:** `.claude/skills/arch-tidy/` — this path is **gitignored** (`.claude/*`), so the skill is **local-only**. Therefore each task's checkpoint is **tests green**, NOT a git commit. The only git-tracked artifacts are the spec and this plan (already/also under `docs/`).
- **`analyze.py` is stdlib-only and read-only** — no third-party imports, no filesystem mutation, no network. All mutations (moves, doc writes) are the agent's job, per `SKILL.md`.
- **Tests are `unittest`** (no pytest dependency). Run with: `cd .claude/skills/arch-tidy && python3 -m unittest discover tests`.
- **Docs are selective** — a folder qualifies for `_folder_context.md` only if top-level domain module, clear boundary/interface, or deep/complex sub-architecture; omit assets/styles/icons/pure-util/single-component leaves.
- **Moves are propose-then-approve** — the script flags *candidates* with rationale; the agent proposes; the user approves before any `git mv`.
- **Idempotency/freshness** — a folder's doc is regenerated only when its `surface_hash` differs from the `last_reviewed` recorded in its front-matter.
- **`_folder_context.md` front-matter fields (verbatim):** `module`, `owns` (list), `boundary` (optional), `last_reviewed`.

```python
import re
from pathlib import PurePosixPath

KIND_TO_OWNS = {
    "data-model": "models", "ui-component": "components", "service": "services",
    "route": "routes", "util": "utils", "style": "styles", "asset": "assets",
    "config": "config", "test": "tests", "doc": "docs",
}


def infer_kind(path: str) -> str:
    p = PurePosixPath(path.replace("\\", "/"))
    name = p.name
    stem = p.stem
    ext = p.suffix.lower()
    lname = name.lower()
    if lname.startswith("test_") or lname.endswith("_test.py") or ".test." in lname or ".spec." in lname:
        return "test"
    if ext in {".css", ".scss", ".sass", ".less"}:
        return "style"
    if ext in {".svg", ".png", ".jpg", ".jpeg", ".gif", ".webp", ".ico", ".woff", ".woff2"}:
        return "asset"
    if ext in {".yaml", ".yml", ".toml", ".ini", ".cfg", ".json"} or stem in {"config", "settings"}:
        return "config"
    if ext in {".md", ".rst", ".txt"}:
        return "doc"
    if ext in {".tsx", ".jsx"} or (ext in {".ts", ".js", ".vue", ".svelte"} and stem[:1].isupper()):
        return "ui-component"
    if re.search(r"(^|[_.])model(s)?([_.]|$)", stem.lower()):
        return "data-model"
    if re.search(r"(^|[_.])(route|router|controller|view|api|endpoint)(s)?([_.]|$)", stem.lower()):
        return "route"
    if re.search(r"(^|[_.])(service|repository|gateway|client|dao)(s)?([_.]|$)", stem.lower()):
        return "service"
    if re.search(r"(^|[_.])(util|helper|shared|common)(s)?([_.]|$)", stem.lower()):
        return "util"
    return "unknown"
```

```python
import os

CONTEXT_FILE = "_folder_context.md"


def parse_context_file(path: str) -> dict | None:
    if not os.path.isfile(path):
        return None
    text = open(path, encoding="utf-8").read()
    m = re.match(r"^---\n(.*?)\n---", text, re.DOTALL)
    fm = m.group(1) if m else ""
    meta = {"module": None, "owns": [], "boundary": None, "last_reviewed": None, "path": path}
    for line in fm.splitlines():
        if ":" not in line:
            continue
        key, _, val = line.partition(":")
        key, val = key.strip(), val.strip()
        if key == "owns":
            meta["owns"] = [x.strip() for x in val.strip("[]").split(",") if x.strip()]
        elif key in meta:
            meta[key] = val or None
    return meta


def nearest_context(file_path: str, root: str) -> str | None:
    d = os.path.dirname(os.path.abspath(file_path))
    root = os.path.abspath(root)
    while d.startswith(root):
        candidate = os.path.join(d, CONTEXT_FILE)
        if os.path.isfile(candidate):
            return candidate
        if d == root:
            break
        d = os.path.dirname(d)
    return None
```

def placement_candidates(root: str, changed_files: list[str]) -> list[dict]:
    out = []
    for rel in changed_files:
        abspath = os.path.join(root, rel)
        near = nearest_context(abspath, root)
        meta = parse_context_file(near) if near else None
        kind = infer_kind(rel)
        category = KIND_TO_OWNS.get(kind)
        conflict = bool(meta and meta["owns"] and category and category not in meta["owns"])
        out.append({
            "file": rel,
            "nearest_context": os.path.relpath(near, root) if near else None,
            "owns": meta["owns"] if meta else [],
            "inferred_kind": kind,
            "category": category,
            "conflict": conflict,
            "rationale": (
                f"{kind} ('{category}') is not in the owning folder's owns={meta['owns']}"
                if conflict else "location consistent with the nearest owns declaration"
            ),
        })
    return out
```

_IMPORT_RE = re.compile(
    r"""(?:^|\n)\s*(?:from\s+([\w./]+)\s+import|import\s+([\w./]+)|"""
    r"""(?:import|require)\s*\(?\s*['"]([^'"]+)['"])""",
    re.MULTILINE,
)


def _module_to_dir(root: str, token: str) -> str | None:
    # Best-effort: turn "src.auth" / "src/auth" / "@/src/auth" into a dir path if it exists.
    tok = token.lstrip("@").lstrip("./").replace(".", "/")
    d = os.path.join(root, tok)
    if os.path.isdir(d):
        return tok
    parent = os.path.dirname(tok)
    if parent and os.path.isdir(os.path.join(root, parent)):
        return parent
    return None


def build_import_graph(root: str, files: list[str]) -> dict:
    fan_in: dict[str, int] = {}
    fan_out: dict[str, int] = {}
    fixup_index: dict[str, list[str]] = {}
    for rel in files:
        abspath = os.path.join(root, rel)
        if not os.path.isfile(abspath):
            continue
        try:
            text = open(abspath, encoding="utf-8", errors="ignore").read()
        except OSError:
            continue
        src_dir = os.path.dirname(rel) or "."
        for m in _IMPORT_RE.finditer(text):
            token = m.group(1) or m.group(2) or m.group(3)
            if not token:
                continue
            target = _module_to_dir(root, token)
            if not target or target == src_dir:
                continue
            fan_in[target] = fan_in.get(target, 0) + 1
            fan_out[src_dir] = fan_out.get(src_dir, 0) + 1
            fixup_index.setdefault(token, []).append(rel)
    return {"fan_in": fan_in, "fan_out": fan_out, "fixup_index": fixup_index}
```

import hashlib


def surface_hash(dir_path: str) -> str:
    try:
        names = sorted(n for n in os.listdir(dir_path) if not n.startswith("."))
    except OSError:
        names = []
    return hashlib.sha1("\n".join(names).encode()).hexdigest()[:12]


def dir_signals(root: str, dir_path: str, import_graph: dict) -> dict:
    abspath = os.path.join(root, dir_path)
    entries = os.listdir(abspath) if os.path.isdir(abspath) else []
    files = [e for e in entries if os.path.isfile(os.path.join(abspath, e)) and e != CONTEXT_FILE]
    subdirs = [e for e in entries if os.path.isdir(os.path.join(abspath, e)) and not e.startswith(".")]
    ctx_path = os.path.join(abspath, CONTEXT_FILE)
    meta = parse_context_file(ctx_path)
    sh = surface_hash(abspath)
    return {
        "path": dir_path,
        "files": len(files),
        "subdirs": len(subdirs),
        "depth": dir_path.strip("/").count("/"),
        "top_level": dir_path.strip("/").count("/") <= 1,
        "has_context": meta is not None,
        "fan_in": import_graph["fan_in"].get(dir_path, 0),
        "fan_out": import_graph["fan_out"].get(dir_path, 0),
        "surface_hash": sh,
        "last_reviewed": meta["last_reviewed"] if meta else None,
        "stale": bool(meta and meta["last_reviewed"] != sh),
    }
```

import json

_TRIVIAL_NAMES = {"assets", "styles", "icons", "img", "images", "fonts", "static", "utils", "__pycache__"}


def _is_trivial(dir_path: str, s: dict) -> bool:
    name = dir_path.strip("/").split("/")[-1].lower()
    if name in _TRIVIAL_NAMES:
        return True
    return s["files"] <= 1 and s["subdirs"] == 0  # single-file / single-component leaf


def _is_significant(s: dict) -> bool:
    return s["top_level"] or s["fan_in"] >= 5 or (s["subdirs"] >= 2 and s["files"] >= 3)


def doc_gaps(signals: list[dict]) -> list[dict]:
    gaps = []
    for s in signals:
        if _is_trivial(s["path"], s):
            continue
        if not s["has_context"] and _is_significant(s):
            gaps.append({"path": s["path"], "reason": "significant but no _folder_context.md", "signals": s})
        elif s["has_context"] and s["stale"]:
            gaps.append({"path": s["path"], "reason": "context file stale vs surface", "signals": s})
    return gaps


def _dirs_in_scope(root: str, scope: list[str]) -> list[str]:
    dirs = set()
    for rel in scope:
        d = os.path.dirname(rel)
        while True:
            dirs.add(d or ".")
            if not d:
                break
            d = os.path.dirname(d)
    return sorted(x for x in dirs if x != "." and os.path.isdir(os.path.join(root, x)))


def analyze(root: str, scope: list[str]) -> dict:
    graph = build_import_graph(root, scope)
    dirs = _dirs_in_scope(root, scope)
    signals = [dir_signals(root, d, graph) for d in dirs]
    return {
        "scope": scope,
        "dirs": signals,
        "placement_candidates": placement_candidates(root, scope),
        "doc_gaps": doc_gaps(signals),
        "import_fixups": graph["fixup_index"],
    }


def main() -> None:
    import argparse
    ap = argparse.ArgumentParser(prog="arch-tidy analyze")
    ap.add_argument("--root", default=".")
    ap.add_argument("paths", nargs="*", help="scope files; default = git diff --name-only")
    args = ap.parse_args()
    scope = args.paths
    if not scope:
        import subprocess
        out = subprocess.run(["git", "-C", args.root, "diff", "--name-only"],
                             capture_output=True, text=True)
        scope = [p for p in out.stdout.splitlines() if p.strip()]
    print(json.dumps(analyze(args.root, scope), indent=2))


if __name__ == "__main__":
    main()
```

- **Front-matter:** `name: arch-tidy`, `description:` (on-demand architecture cleanup + selective folder docs; triggers: "tidy architecture", "update folder docs", "check file placement").
- **When to use / not use.**
- **Inputs & modes:** default scope = working-tree diff; `--docs-only`, `--moves-only`; explicit paths.
